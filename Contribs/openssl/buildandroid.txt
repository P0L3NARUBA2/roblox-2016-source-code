export NDK=$HOME/android-ndk-r10e
export SYSROOT=$NDK/platforms/android-21/arch-arm
export TOOLCHAIN=$NDK/toolchains/arm-linux-androideabi-4.9/prebuilt/linux-x86_64
export PATH=$TOOLCHAIN/bin:$PATH
 
./Configure android-armv7 \
    --prefix=/mnt/h/Contribs/openssl/install \
    --openssldir=/mnt/h/Contribs/openssl/ssl \
    --sysroot=$SYSROOT \
    no-shared \
    no-ssl2 \
    no-ssl3 \
    no-comp \
    no-hw \
    no-engine \
    -D__ANDROID_API__=21 -DOPENSSL_NO_UI_CONSOLE \
    -march=armv7-a \
    -mfloat-abi=softfp \
    -mfpu=neon \
    -mthumb
 
make clean
make depend
make -j$(nproc)
make install
